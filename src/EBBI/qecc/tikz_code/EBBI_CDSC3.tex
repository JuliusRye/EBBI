
% Define colors for Clifford deformations
\newcommand{\setmycolor}[1]{%
    \ifcase#1\relax
        \def\mycolor{white}%      0
    \or \def\mycolor{LimeGreen}%  1
    \or \def\mycolor{BrickRed}%   2
    \or \def\mycolor{Cerulean}%   3
    \or \def\mycolor{Fuchsia}%    4

    \fi
